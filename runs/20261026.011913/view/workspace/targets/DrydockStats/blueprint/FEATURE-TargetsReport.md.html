FEATURE: Targets Report

FieldValue
Version20261025 V1
DescriptionWrite one CSV row per Target, ordered by Target name, followed by a TOTAL row with cost and minutes.
Depends OnFEATURE-SummaryReport.md
Providesddkstat targets
Consumesreport.FilterTarget, report.FilterDates, shared CSV, minutes, and cost formatters

Purpose

ddkstat targets <log> shows what each Drydock Target consumed and cost.

Output

Header, exactly:

target,calls,failed,input_tokens,cached_tokens,output_tokens,cost_usd,minutes
  • One row per distinct job.target among filtered records, ordered ascending by byte value.
  • failed counts records whose status is not succeeded.
  • Token columns are sums; cost_usd is the summed cost written with %.4f; minutes follows the shared minutes rule over the group's summed elapsed_ms.
  • The final row is TOTAL with every column summed over all filtered records.
  • With no matching record: the header and TOTAL,0,0,0,0,0,0.0000,0.0.

import csv, io, json, os, tempfile from collections import Counter from drydock_go import cli_runner

def rec(i, target="T1", day="2026-10-14", status="succeeded", cmd="build", llm="claude", model="m", inp=10, cached=20, out=30, ms=60000, cost=0.25, ts=None): return {"execution_id": f"id-{i}", "started_at": ts or f"{day}T01:02:03.004Z", "status": status, "job": {"command_name": cmd, "target": target, "llm": llm, "model": model}, "result": {"returncode": 0 if status == "succeeded" else 1, "timed_out": False, "stats": {"input_tokens": inp, "cached_input_tokens": cached, "output_tokens": out, "elapsed_ms": ms, "cost_usd": cost}}}

def run(lines, *args): with tempfile.TemporaryDirectory() as d: p = os.path.join(d, "llm.jsonl") with open(p, "w", encoding="utf-8") as f: for l in lines: f.write((l if isinstance(l, str) else json.dumps(l)) + "\n") code, out, err = cli_runner(*args, p) assert code == 0 print(out) print(err) return code, out

names = ["beta", "Alpha", "alpha", "Zed", "Alpha", "beta", "beta"] recs = [rec(i, t) for i, t in enumerate(names)] code, out = run(recs, "targets") assert code == 0 rows = list(csv.reader(io.StringIO(out))) assert rows[0] == ["target", "calls", "failed", "input_tokens", "cached_tokens", "output_tokens", "cost_usd", "minutes"] counts = Counter(names) assert [r[0] for r in rows[1:-1]] == sorted(counts, key=lambda s: s.encode()) for r in rows[1:-1]: assert int(r[1]) == counts[r[0]] assert rows[-1][0] == "TOTAL" === END AC targets-rows-and-order ===
